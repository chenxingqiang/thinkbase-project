  \ifthenelse{ \fpeval{#3+(#6-floor(#6))*60}>59}
  {\fpeval{#2+1+floor(#6)}:\ifthenelse{\fpeval{#3+(#6-floor(#6)-1)*60}<10}

  }}}}};}
